Shiny App

General Explanation

This app illustrates the Beta Item Response Model (BIRM) with and without response styles (RS). It allows you to visualize how different parameters affect the probability density function of the model. The app is divided into several tabs, each focusing on a specific aspect of the BIRM and response styles.

The probability density funcions show the distribution of responses for a given person and item combination, represented by the indices \(i\) and \(j\). The indexes used in the model are:

Index \(i\): Represents the person index in the model.

Index \(j\): Represents the item index in the model.

Tabs

Standard Beta: Visualizes the probability density function of a standard beta distribution with shape parameters \(m_{ij}\) and \(n_{ij}\). This is helpful for understanding the basic properties of beta distributions.

BIRM: Illustrates the original Beta Item Response Model (BIRM) with person parameters (\(\theta_i\)) and item parameters (\(\delta_j\), \(\tau_j\)). This tab allows you to visualize how these parameters affect the probability density function of the model. This is the base model by Noel & Dauvier (2009) we extended.

ERS only: Shows the BIRM with an additional ERS parameter (\(\eta^{ERS}_{i}\)). This tab allows you to visualize the effect of the ERS parameter on the probability density function.

ARS only: Displays the BIRM with an additional ARS parameter (\(\eta^{ARS}_{i}\)) and effect-coded item wording (\(x_j\)). This tab allows you to visualize how the ARS parameter and item wording affect the probability density function.

BIRM-RS: Combines the ERS and ARS parameters in the BIRM, allowing you to visualize both two response styles and their effects on the probability density function.

BIRM-RS Simulation: Simulates data based on the BIRM-RS model. It generates random values for person parameters (\(\theta_i\), \(\eta^{ERS}_{i}\), \(\eta^{ARS}_{i}\)) and item parameters (\(\delta_j\), \(\tau_j\)), and visualizes the resulting density function from the simulated data. This is useful for understanding how the model behaves in a sample.

App

beta_normal <- function(m = 2, n = 2) {
  vals <- dbeta(seq(0, 1, by = 0.01), m, n)
  plot(seq(0, 1, by = 0.01), vals, type = "l", xlim = c(0, 1), 
       xlab = expression(bold(Y[ij])), 
       ylab = expression(bold(f(Y[ij]))), 
       main = bquote(bold("Probability Density Function for B(" * .(round(m, 2)) * ", " * .(round(n, 2)) * ")")))
}

BIRM <- function(theta, delta, tau, equation = FALSE) {
  middle <- theta - delta
  enumerator_m <- middle + tau
  enumerator_n <- -middle + tau
  m <- exp(enumerator_m / 2)
  n <- exp(enumerator_n / 2)
  if (equation == TRUE) return(round(c(m, n), 2))
  beta_normal(m, n)
}

BIRM_ERS <- function(theta, delta, tau, ers, equation = FALSE) {
  middle <- theta - delta
  ers_exp <- exp(ers)
  enumerator_m <- ers_exp * middle + tau
  enumerator_n <- -ers_exp * middle + tau
  m <- exp(enumerator_m / 2)
  n <- exp(enumerator_n / 2)
  if (equation == TRUE) return(round(c(m, n), 2))
  beta_normal(m, n)
}

BIRM_ARS <- function(theta, delta, tau, ars, x = 1, equation = FALSE) {
  x <- as.numeric(x)
  middle <- theta - delta + x * ars
  enumerator_m <- middle + tau
  enumerator_n <- -middle + tau
  m <- exp(enumerator_m / 2)
  n <- exp(enumerator_n / 2)
  if (equation == TRUE) return(round(c(m, n), 2))
  beta_normal(m, n)
}

BIRM_RS <- function(theta, delta, tau, ers, ars, x = 1, equation = FALSE) {
  x <- as.numeric(x)
  middle <- theta - delta + x * ars
  ers_exp <- exp(ers)
  enumerator_m <- ers_exp * middle + tau
  enumerator_n <- -ers_exp * middle + tau
  m <- exp(enumerator_m / 2)
  n <- exp(enumerator_n / 2)
  if (equation == TRUE) return(round(c(m, n), 2))
  beta_normal(m, n)
}

beta_sample <- function(n = 1000, delta = 0, theta_mu = 0, theta_sd = 1, tau = 2, ars_mu = 0, ars_sd = 0.5, x = 1, ers_mu = 0, ers_sd = 0.5) {
  x <- as.numeric(x)
  theta_vals <- rnorm(n, theta_mu, theta_sd)
  ars_vals <- rnorm(n, ars_mu, ars_sd)
  ers_vals <- rnorm(n, ers_mu, ers_sd)

  # do the above vectorized
  tmp <- sapply(1:n, function(i) BIRM_RS(theta = theta_vals[i], delta = delta, tau = tau, ers = ers_vals[i], ars = ars_vals[i], x = x, equation = TRUE)) |>
    t() |> as.data.frame()
  
  vec <- rbeta(n, tmp[, 1], tmp[, 2])

  par(mar=c(5.1,4.5,4.1,2.1))

  plot(density(vec), main = paste0("Density From Simulated Data (n = ", n, ")"), xlab = expression(bold(Y[ij])), ylab = expression(bold(f(hat(Y)[ij]))))
}

server <- function(input, output) {
  output$beta_normal <- renderPlot({
    beta_normal(m = input$m_beta, n = input$n_beta)
  })
  
  output$BIRM <- renderPlot({
    BIRM(theta = input$theta_birm, delta = input$delta_birm, tau = input$tau_birm)
  })
  
  # let us do equation1 and equation2 for BIRM and then delete the old equation
  output$BIRM_equation_title1 <- renderUI({
    tags$div(
      style = "font-weight: bold; font-size: 14px; font-family: sans-serif; text-align: center;",
      tags$b("Equations for the Probability Density Function")
    )
  })

  output$BIRM_equation_title2 <- renderUI({
    tags$div(
      style = "font-weight: bold; font-size: 14px; font-family: sans-serif; text-align: center;",
      tags$b("Equations With Values")
    )
  })

  output$BIRM_equation1 <- renderUI({
    withMathJax(
      paste0(
        "$$m_{ij} = exp(\\frac{\\theta_i - \\delta_j + \\tau_j}{2}),$$\n",
        "$$n_{ij} = exp(\\frac{-(\\theta_i - \\delta_j) + \\tau_j}{2})$$\n"
      )
    )
  })

  output$BIRM_equation2 <- renderUI({
    withMathJax(
      paste0(
        "$$m_{ij} = exp(\\frac{", input$theta_birm, " - ", input$delta_birm, " + ", input$tau_birm, "}{2}) = ", 
        BIRM(theta = input$theta_birm, delta = input$delta_birm, tau = input$tau_birm, equation = TRUE)[1], ",$$\n",
        "$$n_{ij} = exp(\\frac{-(", input$theta_birm, " - ", input$delta_birm, ") + ", input$tau_birm, "}{2}) = ",
        BIRM(theta = input$theta_birm, delta = input$delta_birm, tau = input$tau_birm, equation = TRUE)[2], "$$\n"
      )

  output$BIRM_RS_equation_title1 <- renderUI({

# Shiny App

## General Explanation

This app illustrates the Beta Item Response Model (BIRM) with and without response styles (RS). It allows you to visualize how different parameters affect the probability density function of the model. The app is divided into several tabs, each focusing on a specific aspect of the BIRM and response styles.

The probability density funcions show the distribution of responses for a given person and item combination, represented by the indices $i$ and $j$. The indexes used in the model are:

**Index $i$**: Represents the person index in the model.

**Index $j$**: Represents the item index in the model.


## Tabs

**Standard Beta**: Visualizes the probability density function of a standard beta distribution with shape parameters $m_{ij}$ and $n_{ij}$. This is helpful for understanding the basic properties of beta distributions.

**BIRM**: Illustrates the original Beta Item Response Model (BIRM) with person parameters ($\theta_i$) and item parameters ($\delta_j$, $\tau_j$). This tab allows you to visualize how these parameters affect the probability density function of the model. This is the base model by Noel & Dauvier (2009) we extended.

**ERS only**: Shows the BIRM with an additional ERS parameter ($\eta^{ERS}_{i}$). This tab allows you to visualize the effect of the ERS parameter on the probability density function. 

**ARS only**: Displays the BIRM with an additional ARS parameter ($\eta^{ARS}_{i}$) and effect-coded item wording ($x_j$). This tab allows you to visualize how the ARS parameter and item wording affect the probability density function. 

**BIRM-RS**: Combines the ERS and ARS parameters in the BIRM, allowing you to visualize both two response styles and their effects on the probability density function.

**BIRM-RS Simulation**: Simulates data based on the BIRM-RS model. It generates random values for person parameters ($\theta_i$, $\eta^{ERS}_{i}$, $\eta^{ARS}_{i}$) and item parameters ($\delta_j$, $\tau_j$), and visualizes the resulting density function from the simulated data. This is useful for understanding how the model behaves in a sample.

beta_normal <- function(m = 2, n = 2) {
  vals <- dbeta(seq(0, 1, by = 0.01), m, n)
  plot(seq(0, 1, by = 0.01), vals, type = "l", xlim = c(0, 1), 
       xlab = expression(bold(Y[ij])), 
       ylab = expression(bold(f(Y[ij]))), 
       main = bquote(bold("Probability Density Function for B(" * .(round(m, 2)) * ", " * .(round(n, 2)) * ")")))
}

BIRM <- function(theta, delta, tau, equation = FALSE) {
  middle <- theta - delta
  enumerator_m <- middle + tau
  enumerator_n <- -middle + tau
  m <- exp(enumerator_m / 2)
  n <- exp(enumerator_n / 2)
  if (equation == TRUE) return(round(c(m, n), 2))
  beta_normal(m, n)
}

BIRM_ERS <- function(theta, delta, tau, ers, equation = FALSE) {
  middle <- theta - delta
  ers_exp <- exp(ers)
  enumerator_m <- ers_exp * middle + tau
  enumerator_n <- -ers_exp * middle + tau
  m <- exp(enumerator_m / 2)
  n <- exp(enumerator_n / 2)
  if (equation == TRUE) return(round(c(m, n), 2))
  beta_normal(m, n)
}

BIRM_ARS <- function(theta, delta, tau, ars, x = 1, equation = FALSE) {
  x <- as.numeric(x)
  middle <- theta - delta + x * ars
  enumerator_m <- middle + tau
  enumerator_n <- -middle + tau
  m <- exp(enumerator_m / 2)
  n <- exp(enumerator_n / 2)
  if (equation == TRUE) return(round(c(m, n), 2))
  beta_normal(m, n)
}

BIRM_RS <- function(theta, delta, tau, ers, ars, x = 1, equation = FALSE) {
  x <- as.numeric(x)
  middle <- theta - delta + x * ars
  ers_exp <- exp(ers)
  enumerator_m <- ers_exp * middle + tau
  enumerator_n <- -ers_exp * middle + tau
  m <- exp(enumerator_m / 2)
  n <- exp(enumerator_n / 2)
  if (equation == TRUE) return(round(c(m, n), 2))
  beta_normal(m, n)
}

beta_sample <- function(n = 1000, delta = 0, theta_mu = 0, theta_sd = 1, tau = 2, ars_mu = 0, ars_sd = 0.5, x = 1, ers_mu = 0, ers_sd = 0.5) {
  x <- as.numeric(x)
  theta_vals <- rnorm(n, theta_mu, theta_sd)
  ars_vals <- rnorm(n, ars_mu, ars_sd)
  ers_vals <- rnorm(n, ers_mu, ers_sd)

  # do the above vectorized
  tmp <- sapply(1:n, function(i) BIRM_RS(theta = theta_vals[i], delta = delta, tau = tau, ers = ers_vals[i], ars = ars_vals[i], x = x, equation = TRUE)) |>
    t() |> as.data.frame()
  
  vec <- rbeta(n, tmp[, 1], tmp[, 2])

  par(mar=c(5.1,4.5,4.1,2.1))

  plot(density(vec), main = paste0("Density From Simulated Data (n = ", n, ")"), xlab = expression(bold(Y[ij])), ylab = expression(bold(f(hat(Y)[ij]))))
}
